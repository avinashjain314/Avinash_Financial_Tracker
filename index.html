How to Setup Sheets Sync?

  1. In your Google Sheet, click on Extensions > Apps Script.
  2. Delete any code in the editor and copy-paste the code snippet below.
  3. Click Save. Then click Deploy > New Deployment.
  4. Select type: Web App. Set "Execute as": Me. Set "Who has access": Anyone (this allows the client-side fetch to reach the writer).
  5. Click Deploy, authorize the permissions, and copy the **Web App URL**.
  6. Paste the URL into the **Write API Configuration** card above!
AppsScript Code 
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const salarySheetId = "1WLUQCxL1VB0seHd9LAnTWK4P3MgGMaYrbTSrBSvVHNk";
    const expenseSheetId = "1vXGxv98Saw0MfzGOfgB_QBWr9ezMWF1SQyffk2gIyGw";
    
    // Parse Date YYYY-MM-DD
    const parts = data.date.split('-');
    const year = parseInt(parts[0], 10);
    const month = parseInt(parts[1], 10) - 1; // 0-indexed
    const day = parseInt(parts[2], 10);
    
    // Format: dd/mm/yyyy
    const dd = String(day).padStart(2, '0');
    const mm = String(month + 1).padStart(2, '0');
    const formattedDate = dd + '/' + mm + '/' + year;
    
    // Format: mmm-yy (e.g. Aug-26)
    const monthNames = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    const formattedMonthName = monthNames[month] + '-' + String(year).slice(-2);
    
    // Action handler: DELETE row
    if (data.action === "delete") {
      if (data.sheetType === "salary") {
        const ss = SpreadsheetApp.openById(salarySheetId);
        const sheet = ss.getSheetByName("UPDATED_INCOME");
        const lastRow = sheet.getLastRow();
        if (lastRow <= 1) throw new Error("No data in sheet to delete");
        const values = sheet.getRange(1, 1, lastRow, 5).getValues();
        
        // Scan from bottom to top to find the matching row
        for (let i = lastRow - 1; i >= 1; i--) {
          let cellDateStr = "";
          if (values[i][0] instanceof Date) {
            cellDateStr = Utilities.formatDate(values[i][0], Session.getScriptTimeZone(), "dd/MM/yyyy");
          } else {
            cellDateStr = String(values[i][0]).trim();
          }
          const cellStream = String(values[i][2]).trim();
          const cellAmount = Number(values[i][3]);
          
          if (cellDateStr === formattedDate && cellStream === String(data.stream).trim() && cellAmount === Number(data.amount)) {
            sheet.deleteRow(i + 1);
            return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Income deleted successfully!" }))
              .setMimeType(ContentService.MimeType.JSON)
              .setHeader("Access-Control-Allow-Origin", "*");
          }
        }
        throw new Error("Income record not found for deletion");
        
      } else if (data.sheetType === "expense") {
        const ss = SpreadsheetApp.openById(expenseSheetId);
        const sheet = ss.getSheetByName("Looker's Studio_Update");
        const lastRow = sheet.getLastRow();
        if (lastRow <= 1) throw new Error("No data in sheet to delete");
        const values = sheet.getRange(1, 1, lastRow, 6).getValues();
        
        // Scan from bottom to top to find the matching row
        for (let i = lastRow - 1; i >= 1; i--) {
          let cellDateStr = "";
          if (values[i][0] instanceof Date) {
            cellDateStr = Utilities.formatDate(values[i][0], Session.getScriptTimeZone(), "dd/MM/yyyy");
          } else {
            cellDateStr = String(values[i][0]).trim();
          }
          const cellAmount = Number(values[i][2]);
          const cellCategory = String(values[i][3]).trim();
          const cellDesc = String(values[i][4]).trim();
          
          if (cellDateStr === formattedDate && cellAmount === Number(data.amount) && cellCategory === String(data.category).trim() && cellDesc === String(data.description).trim()) {
            sheet.deleteRow(i + 1);
            return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Expense deleted successfully!" }))
              .setMimeType(ContentService.MimeType.JSON)
              .setHeader("Access-Control-Allow-Origin", "*");
          }
        }
        throw new Error("Expense record not found for deletion");
      }
    }
    
    // Action handler: INSERT row (default)
    if (data.sheetType === "salary") {
      const ss = SpreadsheetApp.openById(salarySheetId);
      const sheet = ss.getSheetByName("UPDATED_INCOME");
      
      // Target Schema: Date | Month | Income Stream | Amount | Month_Date
      const firstDayOfMonthDate = '01/' + mm + '/' + year;
      sheet.appendRow([
        formattedDate,
        formattedMonthName,
        data.stream,
        Number(data.amount),
        firstDayOfMonthDate
      ]);
      
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Income recorded successfully!" }))
        .setMimeType(ContentService.MimeType.JSON)
        .setHeader("Access-Control-Allow-Origin", "*");
        
    } else if (data.sheetType === "expense") {
      const ss = SpreadsheetApp.openById(expenseSheetId);
      const sheet = ss.getSheetByName("Looker's Studio_Update");
      
      // Target Schema: Date | Month | Expense | Reason | Description | Essential/Non
      sheet.appendRow([
        formattedDate,
        formattedMonthName,
        Number(data.amount),
        data.category,
        data.description,
        data.type
      ]);
      
      return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "Expense recorded successfully!" }))
        .setMimeType(ContentService.MimeType.JSON)
        .setHeader("Access-Control-Allow-Origin", "*");
    }
    
    throw new Error("Invalid sheet type: " + data.sheetType);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON)
      .setHeader("Access-Control-Allow-Origin", "*");
  }
}

function doOptions(e) {
  return ContentService.createTextOutput("")
    .setHeader("Access-Control-Allow-Origin", "*")
    .setHeader("Access-Control-Allow-Methods", "POST, GET, OPTIONS")
    .setHeader("Access-Control-Allow-Headers", "Content-Type");
}
